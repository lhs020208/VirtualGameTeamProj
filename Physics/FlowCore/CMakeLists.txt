cmake_minimum_required(VERSION 3.20)
project(FlowCore VERSION 0.2.0 LANGUAGES CXX)
add_library(flow_core src/contract.cpp src/solver.cpp)
target_include_directories(flow_core PUBLIC include)
target_compile_features(flow_core PUBLIC cxx_std_17)
add_executable(flow_contract app/main.cpp)
target_link_libraries(flow_contract PRIVATE flow_core)
add_executable(flow_run app/run.cpp)
target_link_libraries(flow_run PRIVATE flow_core)
# Reuse the existing DEM translation implementation without changing its source.
add_library(flow_particle_motion ../PowderCore/src/core.cpp)
target_include_directories(flow_particle_motion PUBLIC ../PowderCore/include)
target_compile_features(flow_particle_motion PUBLIC cxx_std_17)
add_library(flow_one_way src/one_way.cpp)
target_link_libraries(flow_one_way PUBLIC flow_core flow_particle_motion)
add_executable(flow_particle_run app/particle_run.cpp)
target_link_libraries(flow_particle_run PRIVATE flow_one_way)
add_library(flow_two_way src/two_way.cpp)
target_link_libraries(flow_two_way PUBLIC flow_one_way)
add_executable(flow_two_way_run app/two_way_run.cpp)
target_link_libraries(flow_two_way_run PRIVATE flow_two_way)
add_library(flow_channel src/channel.cpp)
target_link_libraries(flow_channel PUBLIC flow_two_way)
add_executable(flow_channel_run app/channel_run.cpp)
target_link_libraries(flow_channel_run PRIVATE flow_channel)
include(CTest)
if(BUILD_TESTING)
 add_executable(flow_channel_tests tests/channel_tests.cpp)
 target_link_libraries(flow_channel_tests PRIVATE flow_channel)
 add_test(NAME flow_channel COMMAND flow_channel_tests "${CMAKE_CURRENT_BINARY_DIR}/channel_metrics.csv")
 add_executable(flow_two_way_tests tests/two_way_tests.cpp)
 target_link_libraries(flow_two_way_tests PRIVATE flow_two_way)
 add_test(NAME flow_two_way COMMAND flow_two_way_tests "${CMAKE_CURRENT_BINARY_DIR}/two_way_metrics.csv")
 add_executable(flow_one_way_tests tests/one_way_tests.cpp)
 target_link_libraries(flow_one_way_tests PRIVATE flow_one_way)
 add_test(NAME flow_one_way COMMAND flow_one_way_tests "${CMAKE_CURRENT_BINARY_DIR}/one_way_metrics.csv")
 add_executable(flow_particle_motion_tests ../PowderCore/tests/core_tests.cpp)
 target_link_libraries(flow_particle_motion_tests PRIVATE flow_particle_motion)
 add_test(NAME particle_motion_regression COMMAND flow_particle_motion_tests)
 add_executable(flow_solver_tests tests/solver_tests.cpp)
 target_link_libraries(flow_solver_tests PRIVATE flow_core)
 add_test(NAME flow_solver COMMAND flow_solver_tests "${CMAKE_CURRENT_BINARY_DIR}/flow_solver_metrics.csv")
 add_executable(flow_contract_tests tests/contract_tests.cpp)
 target_link_libraries(flow_contract_tests PRIVATE flow_core)
 add_test(NAME flow_contract COMMAND flow_contract_tests)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME flow_channel_cli COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/channel_cli_tests.py" "$<TARGET_FILE:flow_channel_run>" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/buffered_channel.flow" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/buffered.channel")
 add_test(NAME flow_two_way_cli COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/two_way_cli_tests.py" "$<TARGET_FILE:flow_two_way_run>" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/two_way_channel.flow")
 add_test(NAME flow_particle_cli COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/particle_cli_tests.py" "$<TARGET_FILE:flow_particle_run>" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/one_way_channel.flow")
 add_test(NAME flow_run_cli COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_cli_tests.py" "$<TARGET_FILE:flow_run>" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
 add_test(NAME flow_contract_cli COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py" "$<TARGET_FILE:flow_contract>" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
endif()
foreach(target flow_channel_run flow_channel flow_channel_tests flow_core flow_contract flow_contract_tests flow_solver_tests flow_run flow_particle_motion flow_one_way flow_one_way_tests flow_particle_motion_tests flow_particle_run flow_two_way flow_two_way_tests flow_two_way_run)
 if(TARGET ${target})
  if(MSVC)
   target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
  else()
   target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math)
  endif()
 endif()
endforeach()
