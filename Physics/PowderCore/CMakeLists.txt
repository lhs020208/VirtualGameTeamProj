cmake_minimum_required(VERSION 3.20)
project(PowderCore VERSION 0.21.0 LANGUAGES CXX)
add_library(powder_core src/core.cpp src/friction.cpp src/elastic.cpp src/world.cpp src/geometry.cpp src/worker.cpp src/tracking.cpp src/settings.cpp src/supply.cpp src/results.cpp src/clump.cpp src/mesh.cpp src/motion.cpp)
target_include_directories(powder_core PUBLIC include)
target_compile_features(powder_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(powder_core PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(powder_core PRIVATE -Wall -Wextra -Werror -fno-fast-math)
endif()
add_executable(powder_run app/main.cpp)
target_link_libraries(powder_run PRIVATE powder_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(powder_tests tests/core_tests.cpp)
  target_link_libraries(powder_tests PRIVATE powder_core)
  add_test(NAME core_contract COMMAND powder_tests)
endif()
