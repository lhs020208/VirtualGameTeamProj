cmake_minimum_required(VERSION 3.20)
project(PowderCore VERSION 0.7.0 LANGUAGES CXX)
add_library(powder_core src/core.cpp src/friction.cpp src/elastic.cpp src/world.cpp)
target_include_directories(powder_core PUBLIC include)
target_compile_features(powder_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(powder_core PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(powder_core PRIVATE -Wall -Wextra -Werror -fno-fast-math)
endif()
add_executable(powder_run app/main.cpp)
target_link_libraries(powder_run PRIVATE powder_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(powder_tests tests/core_tests.cpp)
  target_link_libraries(powder_tests PRIVATE powder_core)
  add_test(NAME core_contract COMMAND powder_tests)
endif()

if(BUILD_TESTING)
  add_executable(contact_tests tests/contact_tests.cpp)
  target_link_libraries(contact_tests PRIVATE powder_core)
  add_test(NAME plane_contact COMMAND contact_tests)
endif()

if(BUILD_TESTING)
  add_executable(pair_tests tests/pair_tests.cpp)
  target_link_libraries(pair_tests PRIVATE powder_core)
  add_test(NAME sphere_pair COMMAND pair_tests)
endif()

if(BUILD_TESTING)
 add_executable(friction_tests tests/friction_tests.cpp)
 target_link_libraries(friction_tests PRIVATE powder_core)

endif()

if(BUILD_TESTING)
 add_executable(hertz_tests tests/hertz_tests.cpp)

endif()

if(BUILD_TESTING)
 add_executable(world_tests tests/world_tests.cpp)
 target_link_libraries(world_tests PRIVATE powder_core)

endif()
